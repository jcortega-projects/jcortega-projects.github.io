Mentor IA: Code Snippets

Illustrative backend snippet for recommendation-driven mentor response planning.

from fastapi import APIRouter
from pydantic import BaseModel

router = APIRouter()

class MentorRequest(BaseModel):
    learner_goal: str
    recent_activity: list[str]

def build_prompt(payload: MentorRequest) -> str:
    history = ", ".join(payload.recent_activity[-3:])
    return (
        f"Goal: {payload.learner_goal}\n"
        f"Recent activity: {history}\n"
        "Return a concise next-step mentoring plan."
    )

@router.post("/mentor/next-step")
def next_step(payload: MentorRequest):
    prompt = build_prompt(payload)
    return {"prompt": prompt, "status": "ready"}
